# Atieh AI — Ready PowerShell Commands

این فایل شامل دستورات پرتکرار برای اجرای پروژه Atieh AI، دیباگ سیستم، اجرای backend و frontend و تست API ها است.

مسیر پروژه:

C:\Users\USER\Documents\GitHub\atieh


# رفتن به پوشه اصلی پروژه
Set-Location "C:\Users\USER\Documents\GitHub\atieh"



# بستن تمام process های قبلی پروژه (uvicorn / node / npm)
Get-Process uvicorn,node,npm -ErrorAction SilentlyContinue | Stop-Process -Force



# بررسی اینکه آیا پورت backend (8000) اشغال شده یا نه
Get-NetTCPConnection -LocalPort 8000 -ErrorAction SilentlyContinue | Format-Table -AutoSize



# بررسی اینکه آیا پورت frontend (5175) اشغال شده یا نه
Get-NetTCPConnection -LocalPort 5175 -ErrorAction SilentlyContinue | Format-Table -AutoSize



# اجرای backend با FastAPI و Uvicorn
python -m uvicorn main:app --reload --port 8000



# تست سلامت backend
Invoke-RestMethod http://127.0.0.1:8000/health



# رفتن به پوشه frontend
Set-Location "C:\Users\USER\Documents\GitHub\atieh\frontend"



# نصب پکیج های frontend (فقط بار اول)
npm install



# اجرای frontend با Vite
npm run dev



# تست موتور پیشنهاد نوبت AI
Invoke-RestMethod `
-Uri "http://127.0.0.1:8000/ai/engine/recommend-slot" `
-Method POST `
-ContentType "application/json" `
-Body '{
"record_no":139990,
"service":"TREATMENT_1",
"insurance":"CASH",
"preferred_day":"Monday"
}'



# اجرای سیستم import برای پردازش فایل های Excel / CSV
Invoke-RestMethod -Uri "http://127.0.0.1:8000/imports/run" -Method POST



# بررسی فایل های import نشده
Get-ChildItem "C:\AtiehAI\incoming"



# بررسی فایل های پردازش شده
Get-ChildItem "C:\AtiehAI\processed"



# ایجاد یک فایل تست برای سیستم import
"test_col`n123" | Set-Content "C:\AtiehAI\incoming\test_import.csv" -Encoding UTF8



# بررسی ساختار دیتابیس SQLite
python inspect_schema.py
